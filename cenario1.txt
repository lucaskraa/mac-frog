function currentAreaRoom() {
  return currentRoom === 0 ? AREA1.room0 : AREA1.room1;
}

function cloneGeometry(items) {
  return items.map(o => {
    const c = { ...o };
    delete c.style;
    return c;
  });
}

function findAreaAnchor() {
  const data = currentAreaRoom();
  const origin = getTongueOrigin();
  let best = null;
  let bestScore = Infinity;
  for (const anchor of data.anchors) {
    const d = distance(origin.x, origin.y, anchor.x, anchor.y);
    if (d > anchor.grabRange) continue;
    if (!tongueLineClearToPoint(anchor.x, anchor.y)) continue;
    const direction = anchor.x >= origin.x ? 1 : -1;
    if (direction !== mac.facing && d > 180) continue;
    const score = d + (anchor.y < origin.y ? -28 : 0);
    if (score < bestScore) {
      best = anchor;
      bestScore = score;
    }
  }
  return best;
}

canGrabPoint = function() {
  if (grapple.active) return true;
  const anchor = findAreaAnchor();
  if (!anchor) return false;
  grapplePoint.x = anchor.x;
  grapplePoint.y = anchor.y;
  grapplePoint.radius = anchor.radius;
  grapplePoint.grabRange = anchor.grabRange;
  return true;
};

rebuildSolids = function() {
  solids.length = 0;
  const data = currentAreaRoom();
  solids.push(...data.floors, ...platforms, ...walls);
  if (boss.active && boss.alive) solids.push(bossGate);
};

function makeMangueEnemy(room, type, x, patrolMin, patrolMax, opts = {}) {
  const profile = {
    caranguejo: { w: 48, h: 34, hp: 78, patrol: 72, chase: 126, damage: 9 },
    cururu:     { w: 56, h: 48, hp: 130, patrol: 42, chase: 82, damage: 15 },
    perereca:   { w: 42, h: 42, hp: 92, patrol: 58, chase: 108, damage: 11 }
  }[type] || { w: 46, h: 42, hp: 90, patrol: 55, chase: 105, damage: 10 };
  return {
    room, type, spawnX: x, x, y: 0,
    surfaceY: opts.surfaceY ?? ground.y,
    width: opts.width ?? profile.w,
    height: opts.height ?? profile.h,
    maxHp: opts.maxHp ?? profile.hp,
    hp: opts.maxHp ?? profile.hp,
    alive: true,
    patrolMin, patrolMax,
    patrolSpeed: opts.patrolSpeed ?? profile.patrol,
    chaseSpeed: opts.chaseSpeed ?? profile.chase,
    detectionRange: opts.detectionRange ?? 285,
    direction: opts.direction ?? 1,
    damage: opts.damage ?? profile.damage,
    attackCooldown: opts.attackCooldown ?? 0.85,
    attackCooldownTimer: 0,
    hurtTimer: 0,
    hurtDuration: 0.12,
    animTime: 0,
    phase: opts.phase ?? 0
  };
}

function getExtraEnemies() {
  return mangueEnemies.filter(e => e.room === currentRoom);
}

function resetMangueEnemies(room) {
  for (const e of mangueEnemies) {
    if (e.room !== room) continue;
    e.x = e.spawnX;
    e.y = e.surfaceY - e.height;
    e.hp = e.maxHp;
    e.alive = true;
    e.attackCooldownTimer = 0;
    e.hurtTimer = 0;
    e.animTime = 0;
  }
}

function updateOneMangueEnemy(e, dt) {
  if (!e.alive) return;
  e.animTime = (e.animTime || 0) + dt;
  if (e.hurtTimer > 0) e.hurtTimer -= dt;
  if (e.attackCooldownTimer > 0) e.attackCooldownTimer -= dt;

  const bob = e.type === "perereca"
    ? Math.max(0, Math.sin(e.animTime * 3.2 + (e.phase || 0))) * 7
    : 0;
  e.y = (e.surfaceY ?? ground.y) - e.height - bob;

  const ec = e.x + e.width / 2;
  const mc = mac.x + mac.width / 2;
  const dx = Math.abs(mc - ec);
  const dy = Math.abs((mac.y + mac.height / 2) - (e.y + e.height / 2));
  let speed = e.patrolSpeed;

  if (!mac.dead && dx <= e.detectionRange && dy < 100) {
    e.direction = mc < ec ? -1 : 1;
    speed = e.chaseSpeed;
  } else {
    if (e.x <= e.patrolMin) e.direction = 1;
    if (e.x + e.width >= e.patrolMax) e.direction = -1;
  }

  e.x += e.direction * speed * dt;
  e.x = Math.max(e.patrolMin, Math.min(e.x, e.patrolMax - e.width));

  if (!mac.dead &&
      rectsOverlap({ x:e.x, y:e.y, width:e.width, height:e.height }, getMacHitbox()) &&
      e.attackCooldownTimer <= 0) {
    damageMac(e.damage, e.x + e.width / 2);
    e.attackCooldownTimer = e.attackCooldown;
  }
}

updateEnemy = function(dt) {
  if (enemy.alive) {
    if (typeof enemy.type !== "string") enemy.type = "caranguejo";
    if (typeof enemy.surfaceY !== "number") enemy.surfaceY = ground.y;
    if (typeof enemy.animTime !== "number") enemy.animTime = 0;
    updateOneMangueEnemy(enemy, dt);
  }
  for (const e of getExtraEnemies()) updateOneMangueEnemy(e, dt);
};

function damageExtraEnemy(e, amount) {
  if (!e || !e.alive) return;
  e.hp -= amount;
  e.hurtTimer = e.hurtDuration;
  if (e.hp <= 0) {
    e.hp = 0;
    e.alive = false;
  }
}

function hitExtraEnemy(hitbox, amount) {
  if (!hitbox) return false;
  for (const e of getExtraEnemies()) {
    if (!e.alive) continue;
    if (rectsOverlap(hitbox, { x:e.x, y:e.y, width:e.width, height:e.height })) {
      damageExtraEnemy(e, amount);
      return true;
    }
  }
  return false;
}

function updateExtraEnemyCombat() {
  if (tongue.active && !grapple.active && !tongue.didHitEnemy && hitExtraEnemy(getTongueHitbox(), 15)) tongue.didHitEnemy = true;
  if (normalAttack.active && normalAttack.hitboxActive && !normalAttack.didHitEnemy && hitExtraEnemy(getNormalAttackHitbox(), 25)) normalAttack.didHitEnemy = true;

  for (let i = waterShots.length - 1; i >= 0; i--) {
    const s = waterShots[i];
    if (hitExtraEnemy({ x:s.x-s.radius, y:s.y-s.radius, width:s.radius*2, height:s.radius*2 }, 20)) waterShots.splice(i,1);
  }

  if (waterBurst.active && !waterBurst.didHitEnemy && hitExtraEnemy(getWaterBurstHitbox(), 35)) waterBurst.didHitEnemy = true;
  if (waterUltimate.active && !waterUltimate.didHitEnemy &&
      hitExtraEnemy({ x:waterUltimate.waveX, y:ground.y-waterUltimate.waveHeight, width:waterUltimate.waveWidth, height:waterUltimate.waveHeight }, 60)) {
    waterUltimate.didHitEnemy = true;
  }

  for (let i = windBlades.length - 1; i >= 0; i--) {
    const b = windBlades[i];
    if (hitExtraEnemy({ x:b.x, y:b.y, width:b.width, height:b.height }, 20)) windBlades.splice(i,1);
  }

  if (windGust.active && !windGust.didHitEnemy && hitExtraEnemy(getWindGustHitbox(), 35)) windGust.didHitEnemy = true;
  if (windUltimate.active && windUltimate.timer >= windUltimate.spawnDelay && !windUltimate.didHitEnemy &&
      hitExtraEnemy({ x:windUltimate.x, y:ground.y-windUltimate.height, width:windUltimate.width, height:windUltimate.height }, 70)) {
    windUltimate.didHitEnemy = true;
  }
}

const __area1BaseUpdateWaterFly = updateWaterFly;
updateWaterFly = function(dt) {
  if (currentRoom !== 0 || !waterFly.alive || typeof waterFly.baseX !== "number") {
    __area1BaseUpdateWaterFly(dt);
    return;
  }
  waterFly.animTime = (waterFly.animTime || 0) + dt;
  waterFly.challengeTime = (waterFly.challengeTime || 0) + dt;
  const t = waterFly.challengeTime;
  waterFly.x = waterFly.baseX + Math.sin(t * 1.55) * 62;
  waterFly.y = waterFly.baseY + Math.sin(t * 3.1) * 25 + Math.sin(t * 1.2) * 8;
};

const __area1BaseUpdateWindFly = updateWindFly;
updateWindFly = function(dt) {
  if (currentRoom !== 1 || !windFly.alive || typeof windFly.baseX !== "number") {
    __area1BaseUpdateWindFly(dt);
    return;
  }
  windFly.animTime = (windFly.animTime || 0) + dt;
  windFly.challengeTime = (windFly.challengeTime || 0) + dt;
  const t = windFly.challengeTime;
  windFly.x = windFly.baseX + Math.sin(t * 1.35) * 48;

function seededNoise(n) {
  const x = Math.sin(n * 12.9898 + 78.233) * 43758.5453;
  return x - Math.floor(x);
}

function drawCloud(x, y, scale, alpha) {
  ctx.save();
  ctx.globalAlpha = alpha;
  ctx.fillStyle = "#ffffff";
  const parts = [[0,0,36],[35,-12,30],[66,0,38],[92,8,26],[28,12,38],[64,18,40]];
  for (const [px,py,r] of parts) {
    ctx.beginPath();
    ctx.arc(x + px*scale, y + py*scale, r*scale, 0, Math.PI*2);
    ctx.fill();
  }
  ctx.restore();
}

function drawDistantChurch(baseX, baseY, alpha=0.3) {
  ctx.save();
  ctx.globalAlpha = alpha;
  ctx.fillStyle = "#737b69";

  ctx.fillStyle = "rgba(235,238,211,0.35)";

  ctx.restore();
}

function drawDistantArch(x, baseY, scale, alpha) {
  ctx.save();
  ctx.globalAlpha = alpha;
  ctx.fillStyle = "#5d6f5e";
  ctx.fillRect(x, baseY-120*scale, 20*scale, 120*scale);
  ctx.fillRect(x+86*scale, baseY-120*scale, 20*scale, 120*scale);
  ctx.strokeStyle = "#5d6f5e";
  ctx.lineWidth = 18*scale;
  ctx.beginPath();
  ctx.arc(x+53*scale, baseY-112*scale, 44*scale, Math.PI, Math.PI*2);
  ctx.stroke();
  ctx.restore();
}

function drawMangroveBackground() {

  if (currentRoom === 0) {
    sky.addColorStop(0,p.skyTop);
    sky.addColorStop(0.45,p.skyMid);
    sky.addColorStop(1,p.skyLow);
  } else {
    sky.addColorStop(0,"#77aebd");
    sky.addColorStop(0.48,"#a9c5b0");
    sky.addColorStop(1,"#6f8d72");
  }

  ctx.fillStyle = sky;
  ctx.fillRect(0,0,WIDTH,HEIGHT);

  if (currentRoom === 0) {
    drawCloud(90 - (camera.x*0.015)%360, 95, 0.75, 0.36);
    drawCloud(620 - (camera.x*0.02)%500, 145, 0.55, 0.28);
  }

  ctx.save();
  const glow = ctx.createRadialGradient(WIDTH*0.73,105,10,WIDTH*0.73,105,150);
  glow.addColorStop(0,"rgba(255,245,190,0.34)");
  glow.addColorStop(1,"rgba(255,245,190,0)");
  ctx.fillStyle = glow;
  ctx.fillRect(0,0,WIDTH,HEIGHT);
  ctx.restore();

  const far = -((camera.x*0.055)%520);
  ctx.fillStyle = p.farHill;

  for (let i=-1;i<Math.ceil(WIDTH/360)+2;i++) {
    const x = far+i*360;
    ctx.beginPath();
    ctx.moveTo(x,HEIGHT*0.63);
    ctx.quadraticCurveTo(x+85,HEIGHT*0.31,x+190,HEIGHT*0.56);
    ctx.quadraticCurveTo(x+275,HEIGHT*0.38,x+370,HEIGHT*0.63);
    ctx.lineTo(x+370,HEIGHT*0.72);
    ctx.lineTo(x,HEIGHT*0.72);
    ctx.closePath();
    ctx.fill();
  }

  drawDistantChurch(WIDTH*0.70-(camera.x*0.035)%260, 205, currentRoom===0?0.27:0.18);

  const archOffset = -((camera.x*0.09)%480);
  for (let i=-1;i<4;i++) drawDistantArch(archOffset+i*480+170, HEIGHT*0.64, 0.8, 0.22);

  const mid = -((camera.x*0.17)%250);
  ctx.strokeStyle = p.midTree;
  ctx.lineCap = "round";

  for (let i=-1;i<Math.ceil(WIDTH/150)+2;i++) {
    const x = mid+i*150+35;
    const base = HEIGHT*0.70;
    const top = HEIGHT*(0.29+(i%4)*0.025);

    ctx.lineWidth = 8;
    ctx.beginPath();
    ctx.moveTo(x,base);
    ctx.quadraticCurveTo(x-16,(base+top)/2,x,top);
    ctx.stroke();

    ctx.lineWidth = 3.5;
    for (let a=-2;a<=2;a++) {
      ctx.beginPath();
      ctx.moveTo(x,top+8);
      ctx.quadraticCurveTo(x+a*16,top-15,x+a*34,top+10);
      ctx.stroke();
    }
  }

  const waterY = HEIGHT*0.68;
  const waterGrad = ctx.createLinearGradient(0,waterY,0,HEIGHT);
  waterGrad.addColorStop(0,"rgba(76,157,156,0.38)");
  waterGrad.addColorStop(1,"rgba(34,94,91,0.50)");

function platformStyleFor(rect) {
  const source = currentAreaRoom().platforms.find(p => p.x===rect.x && p.y===rect.y && p.width===rect.width);
  return source?.style || "stone";
}

function drawStonePlatform(rect, style="stone") {
  const p = AREA1_PALETTE;

  if (style === "wood") {
    ctx.fillStyle = p.woodDark;
    ctx.fillRect(rect.x,rect.y,rect.width,rect.height);
    ctx.fillStyle = p.wood;
    ctx.fillRect(rect.x,rect.y,rect.width,7);

    ctx.strokeStyle = "rgba(43,28,18,0.55)";
    ctx.lineWidth = 2;
    for (let x=rect.x+22;x<rect.x+rect.width;x+=42) {
      ctx.beginPath();
      ctx.moveTo(x,rect.y);
      ctx.lineTo(x-4,rect.y+rect.height);
      ctx.stroke();
    }

    ctx.strokeStyle = "#435b38";
    for (let x=rect.x+18;x<rect.x+rect.width-12;x+=55) {
      ctx.beginPath();
      ctx.moveTo(x,rect.y+rect.height);
      ctx.quadraticCurveTo(x+4,rect.y+36,x-5,rect.y+58);
      ctx.stroke();
    }
    return;
  }

  const side = style === "shrine" ? "#67654f" : (style === "arena" ? "#5b5c49" : p.stone);
  const top = style === "shrine" ? "#a59a72" : p.stoneTop;
  const grad = ctx.createLinearGradient(0,rect.y,0,rect.y+rect.height);
  grad.addColorStop(0,top);
  grad.addColorStop(0.38,side);
  grad.addColorStop(1,p.stoneDark);

  ctx.fillStyle = grad;
  ctx.fillRect(rect.x,rect.y,rect.width,rect.height);

  ctx.strokeStyle = "rgba(47,49,38,0.44)";
  ctx.lineWidth = 1;

  const bw = 38, bh = 16;
  for (let yy=rect.y+8;yy<rect.y+rect.height;yy+=bh) {
    const row = Math.floor((yy-rect.y)/bh);
    for (let xx=rect.x+(row%2?-19:0);xx<rect.x+rect.width;xx+=bw) {
      ctx.strokeRect(xx,yy,bw,bh);
    }
  }

  ctx.fillStyle = p.moss;
  for (let x=rect.x+5;x<rect.x+rect.width-4;x+=19) {
    const h = 3+Math.floor(seededNoise(x*.2+rect.y)*5);
    ctx.fillRect(x,rect.y-3,14,5);
    if (h>5) ctx.fillRect(x+4,rect.y+1,3,h);
  }

  if (style === "shrine") {
    ctx.strokeStyle = p.azulejo;
    ctx.lineWidth = 2;
    for (let x=rect.x+22;x<rect.x+rect.width-12;x+=44) {
      ctx.beginPath();
      ctx.moveTo(x,rect.y+7);
      ctx.lineTo(x+8,rect.y+13);
      ctx.lineTo(x+16,rect.y+7);
      ctx.lineTo(x+8,rect.y+2);
      ctx.closePath();
      ctx.stroke();
    }
  }
}

}

function drawWorldGeometry() {

  for (const platform of platforms) drawStonePlatform(platform, platformStyleFor(platform));

function drawAnchor(anchor, reachable) {
  const p = AREA1_PALETTE;
  ctx.save();

  const isChain = anchor.kind === "chainroot";
  ctx.strokeStyle = isChain ? "#665f50" : (anchor.kind==="vine" ? p.vine : p.root);
  ctx.lineWidth = isChain ? 5 : (anchor.kind==="vine" ? 6 : 10);
  ctx.lineCap = "round";

  ctx.beginPath();
  ctx.moveTo(anchor.x-(anchor.kind==="root"?52:0),anchor.y-64);
  ctx.quadraticCurveTo(anchor.x-8,anchor.y-29,anchor.x,anchor.y);
  ctx.stroke();

  if (isChain) {
    ctx.strokeStyle = "#a09579";
    ctx.lineWidth = 3;
    for (let y=anchor.y-55;y<anchor.y-3;y+=10) {
      ctx.beginPath();
      ctx.ellipse(anchor.x,y,5,7,0,0,Math.PI*2);
      ctx.stroke();
    }
  }

  ctx.fillStyle = reachable ? "#d6e998" : "#83945f";
  ctx.beginPath();
  ctx.ellipse(anchor.x,anchor.y,11,8,-.2,0,Math.PI*2);
  ctx.fill();

  ctx.strokeStyle = reachable ? "#f1f7c2" : "#c2ce8a";
  ctx.lineWidth=2;
  ctx.stroke();

  if (reachable) {
    ctx.globalAlpha=.22;
    ctx.fillStyle="#eaffbb";
    ctx.beginPath();
    ctx.arc(anchor.x,anchor.y,22+Math.sin(performance.now()*.008)*3,0,Math.PI*2);
    ctx.fill();
  }
  ctx.restore();
}

drawGrapplePoint = function() {
  const origin = getTongueOrigin();
  for (const anchor of currentAreaRoom().anchors) {
    const d = distance(origin.x,origin.y,anchor.x,anchor.y);
    const dir = anchor.x>=origin.x?1:-1;
    const reachable = !grapple.active &&
      d<=anchor.grabRange &&
      (dir===mac.facing || d<=180) &&
      tongueLineClearToPoint(anchor.x,anchor.y);

    drawAnchor(anchor,reachable);
  }
};

function drawMangueEnemy(e) {
  if (!e.alive) return;

  const t = e.animTime || 0;
  const hurt = e.hurtTimer > 0;
  const cx=e.x+e.width/2;
  const cy=e.y+e.height/2;

  ctx.save();
  ctx.lineJoin="round";
  ctx.lineCap="round";

  if (e.type === "caranguejo") {
    const step=Math.sin(t*11)*3;

    ctx.strokeStyle="#5c3027";
    ctx.lineWidth=5;

    for (const d of [-1,1]) {
      ctx.beginPath();
      ctx.moveTo(cx+d*13,cy+7);
      ctx.lineTo(cx+d*(25+step),cy+16);
      ctx.lineTo(cx+d*32,cy+11);
      ctx.stroke();

      ctx.beginPath();
      ctx.moveTo(cx+d*12,cy+1);
      ctx.lineTo(cx+d*(25-step),cy-7);
      ctx.stroke();
    }

    ctx.fillStyle=hurt?"#f2aaa0":"#b95d3e";
    ctx.strokeStyle="#71382c";
    ctx.lineWidth=2.5;
    ctx.beginPath();
    ctx.ellipse(cx,cy+5,e.width*.39,e.height*.29,0,0,Math.PI*2);
    ctx.fill();
    ctx.stroke();

    ctx.fillStyle="#e7865e";
    ctx.beginPath();
    ctx.ellipse(cx,cy+1,e.width*.25,e.height*.16,0,0,Math.PI*2);
    ctx.fill();

    ctx.fillStyle="#f2d9a1";
    ctx.beginPath();
    ctx.arc(cx-8,cy-7,3,0,Math.PI*2);
    ctx.arc(cx+8,cy-7,3,0,Math.PI*2);
    ctx.fill();

    ctx.fillStyle="#22271f";
    ctx.beginPath();
    ctx.arc(cx-8,cy-7,1.2,0,Math.PI*2);
    ctx.arc(cx+8,cy-7,1.2,0,Math.PI*2);
    ctx.fill();
  } else {
    const isPerereca=e.type==="perereca";
    const body = isPerereca ? "#5e8a4c" : "#7c7245";
    const light = isPerereca ? "#8dbb63" : "#9a8e55";
    const outline = isPerereca ? "#355537" : "#514a32";
    const squash = isPerereca ? Math.sin(t*6)*2 : Math.sin(t*3)*1.2;

    ctx.strokeStyle=outline;
    ctx.lineWidth=4;
    ctx.beginPath();
    ctx.moveTo(cx-13,cy+13);
    ctx.lineTo(cx-23,cy+20);
    ctx.moveTo(cx+13,cy+13);
    ctx.lineTo(cx+23,cy+20);
    ctx.stroke();

    ctx.fillStyle=hurt?"#e6aaa2":body;
    ctx.strokeStyle=outline;
    ctx.lineWidth=2.5;
    ctx.beginPath();
    ctx.ellipse(cx,cy+6,e.width*.45,e.height*.34+squash,0,0,Math.PI*2);
    ctx.fill();
    ctx.stroke();

    ctx.fillStyle=hurt?"#f4c2b9":light;
    ctx.beginPath();
    ctx.ellipse(cx,e.y+14,e.width*.33,14,0,0,Math.PI*2);
    ctx.fill();

    ctx.fillStyle="#eadb9e";
    ctx.beginPath();
    ctx.arc(cx-9,e.y+8,4,0,Math.PI*2);
    ctx.arc(cx+9,e.y+8,4,0,Math.PI*2);
    ctx.fill();

    ctx.fillStyle="#1d261d";
    ctx.beginPath();
    ctx.arc(cx-9,e.y+8,1.8,0,Math.PI*2);
    ctx.arc(cx+9,e.y+8,1.8,0,Math.PI*2);
    ctx.fill();
  }

  const ratio=Math.max(0,e.hp/e.maxHp);
  ctx.fillStyle="rgba(27,35,28,.78)";
  ctx.fillRect(e.x,e.y-10,e.width,4);

    ctx.strokeStyle="#ffe082";
    ctx.lineWidth=1;
    ctx.strokeRect(e.x,e.y,e.width,e.height);
  }
  ctx.restore();
}

function v6Clone(items) {
  return items.map(item => {
    const o = { ...item };
    delete o.style;
    return o;
  });
}

function v6PlatformStyle(rect) {
  const p = currentAreaRoom().platforms.find(s => s.x === rect.x && s.y === rect.y && s.width === rect.width);

}

rebuildSolids = function() {
  solids.length = 0;
  const data = currentAreaRoom();
  solids.push(...data.floors, ...platforms, ...walls);
  if (boss.active && boss.alive) solids.push(bossGate);
};

findAreaAnchor = function() {
  const origin = getTongueOrigin();
  let best = null;
  let bestScore = Infinity;
  for (const anchor of currentAreaRoom().anchors) {
    const d = distance(origin.x, origin.y, anchor.x, anchor.y);
    if (d > anchor.grabRange) continue;
    if (!tongueLineClearToPoint(anchor.x, anchor.y)) continue;
    const dir = anchor.x >= origin.x ? 1 : -1;
    if (dir !== mac.facing && d > 175) continue;
    const score = d + (anchor.y < origin.y ? -32 : 0);
    if (score < bestScore) {
      best = anchor;
      bestScore = score;
    }
  }
  return best;
};

canGrabPoint = function() {
  if (grapple.active) return true;
  const anchor = findAreaAnchor();
  if (!anchor) return false;
  grapplePoint.x = anchor.x;
  grapplePoint.y = anchor.y;
  grapplePoint.radius = anchor.radius;
  grapplePoint.grabRange = anchor.grabRange;
  return true;
};

let areaGateMessageTimer = 0;
updateRoomTransition = function(dt) {
  if (areaGateMessageTimer > 0) areaGateMessageTimer -= dt;
  if (roomTransitionCooldown > 0) {
    roomTransitionCooldown -= dt;
    return;
  }
  if (currentRoom === 0 && mac.x >= WORLD_WIDTH - mac.width - 2) {
    if (!mac.waterPower) {
      mac.x = WORLD_WIDTH - mac.width - 70;
      mac.vx = -120;
      areaGateMessageTimer = 1.8;

    }
    loadRoom(1,'left');
    return;
  }
  if (currentRoom === 1 && mac.x <= 2) loadRoom(0,'right');
};

const __v6WaterFly = updateWaterFly;
updateWaterFly = function(dt) {
  if (currentRoom !== 0 || !waterFly.alive) {
    __v6WaterFly(dt);

  waterFly.challengeTime = (waterFly.challengeTime || 0) + dt;

const V6_PIXEL_DIV = 2;
const v6PixelCanvas = document.createElement('canvas');
const v6PixelCtx = v6PixelCanvas.getContext('2d');

function v6EnsurePixelBuffer() {
  const w = Math.max(1, Math.floor(WIDTH / V6_PIXEL_DIV));
  const h = Math.max(1, Math.floor(HEIGHT / V6_PIXEL_DIV));
  if (v6PixelCanvas.width !== w || v6PixelCanvas.height !== h) {
    v6PixelCanvas.width = w;
    v6PixelCanvas.height = h;
  }
  v6PixelCtx.imageSmoothingEnabled = false;
}

function v6PixelateWorld() {
  v6EnsurePixelBuffer();
  v6PixelCtx.clearRect(0,0,v6PixelCanvas.width,v6PixelCanvas.height);
  v6PixelCtx.drawImage(canvas,0,0,WIDTH,HEIGHT,0,0,v6PixelCanvas.width,v6PixelCanvas.height);
  ctx.save();
  ctx.setTransform(1,0,0,1,0,0);
  ctx.imageSmoothingEnabled = false;
  ctx.clearRect(0,0,WIDTH,HEIGHT);
  ctx.drawImage(v6PixelCanvas,0,0,v6PixelCanvas.width,v6PixelCanvas.height,0,0,WIDTH,HEIGHT);
  ctx.restore();
}

function v6Noise(n) {
  const x = Math.sin(n*12.9898 + 78.233)*43758.5453;
  return x-Math.floor(x);
}

function v6DrawEnemy(e) {
  if(!e.alive)return;

    ctx.fillStyle=hurt?'#f0a59a':'#9b4e37';
    ctx.fillRect(x+10,y+12,28,16);
    ctx.fillStyle='#c96746'; ctx.fillRect(x+14,y+8,20,8);

    ctx.fillStyle=outline; ctx.fillRect(x+5,y+16+bounce,e.width-10,e.height-17-bounce);
    ctx.fillStyle=hurt?'#e4a59e':body; ctx.fillRect(x+9,y+12+bounce,e.width-18,e.height-17-bounce);
