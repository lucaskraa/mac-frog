function updateOneMangueEnemy(e, dt) {
  if (!e.alive) return;
  if (e.hurtTimer > 0) e.hurtTimer -= dt;
  if (e.attackCooldownTimer > 0) e.attackCooldownTimer -= dt;
  e.y = ground.y - e.height;
  const enemyCenter = e.x + e.width / 2;
  const macCenter = mac.x + mac.width / 2;
  const distance = Math.abs(macCenter - enemyCenter);
  let speed = e.patrolSpeed;
  if (!mac.dead && distance <= e.detectionRange) {
    e.direction = macCenter < enemyCenter ? -1 : 1;
    speed = e.chaseSpeed;
  } else {
    if (e.x <= e.patrolMin) e.direction = 1;
    if (e.x + e.width >= e.patrolMax) e.direction = -1;
  }
  e.x += e.direction * speed * dt;
  e.x = Math.max(
    e.patrolMin,
    Math.min(e.x, e.patrolMax - e.width)
  );
  if (mac.dead) return;
  const hitbox = {
    x: e.x,
    y: e.y,
    width: e.width,
    height: e.height
  };
  if (
    rectsOverlap(hitbox, getMacHitbox()) &&
    e.attackCooldownTimer <= 0
  ) {
    damageMac(e.damage, e.x + e.width / 2);
    e.attackCooldownTimer = e.attackCooldown;
  }
}
updateEnemy = function(dt) {
  updateOneMangueEnemy(enemy, dt);
  for (const e of getExtraEnemies()) {
    updateOneMangueEnemy(e, dt);
  }
};
function damageExtraEnemy(e, amount) {
  if (!e || !e.alive) return;
  e.hp -= amount;
  e.hurtTimer = e.hurtDuration;
  if (e.hp <= 0) {
    e.hp = 0;
    e.alive = false;
  }
}
function extraEnemyHitbox(e) {
  return {
    x: e.x,
    y: e.y,
    width: e.width,
    height: e.height
  };
}
function hitFirstExtra(hitbox, amount) {
  if (!hitbox) return false;
  for (const e of getExtraEnemies()) {
    if (!e.alive) continue;
    if (rectsOverlap(hitbox, extraEnemyHitbox(e))) {
      damageExtraEnemy(e, amount);
      return true;
    }
  }
  return false;
}
function updateExtraEnemyCombat() {
  if (
    tongue.active &&
    !grapple.active &&
    !tongue.didHitEnemy
  ) {
    if (hitFirstExtra(getTongueHitbox(), 15)) {
      tongue.didHitEnemy = true;
    }
  }
  if (
    normalAttack.active &&
    normalAttack.hitboxActive &&
    !normalAttack.didHitEnemy
  ) {
    if (hitFirstExtra(getNormalAttackHitbox(), 25)) {
      normalAttack.didHitEnemy = true;
    }
  }
  for (let i = waterShots.length - 1; i >= 0; i--) {
    const shot = waterShots[i];
    const hit = {
      x: shot.x - shot.radius,
      y: shot.y - shot.radius,
      width: shot.radius * 2,
      height: shot.radius * 2
    };
    if (hitFirstExtra(hit, 20)) {
      waterShots.splice(i, 1);
    }
  }
  if (
    waterBurst.active &&
    !waterBurst.didHitEnemy
  ) {
    if (hitFirstExtra(getWaterBurstHitbox(), 35)) {
      waterBurst.didHitEnemy = true;
    }
  }
  if (
    waterUltimate.active &&
    !waterUltimate.didHitEnemy
  ) {
    const hit = {
      x: waterUltimate.waveX,
      y: ground.y - waterUltimate.waveHeight,
      width: waterUltimate.waveWidth,
      height: waterUltimate.waveHeight
    };
    if (hitFirstExtra(hit, 60)) {
      waterUltimate.didHitEnemy = true;
    }
  }
  for (let i = windBlades.length - 1; i >= 0; i--) {
    const blade = windBlades[i];
    const hit = {
      x: blade.x,
      y: blade.y,
      width: blade.width,
      height: blade.height
    };
    if (hitFirstExtra(hit, 20)) {
      windBlades.splice(i, 1);
    }
  }
  if (
    windGust.active &&
    !windGust.didHitEnemy
  ) {
    if (hitFirstExtra(getWindGustHitbox(), 35)) {
      windGust.didHitEnemy = true;
    }
  }
  if (
    windUltimate.active &&
    windUltimate.timer >= windUltimate.spawnDelay &&
    !windUltimate.didHitEnemy
  ) {
    const hit = {
      x: windUltimate.x,
      y: ground.y - windUltimate.height,
      width: windUltimate.width,
      height: windUltimate.height
    };
    if (hitFirstExtra(hit, 70)) {
      windUltimate.didHitEnemy = true;
    }
  }
}

function drawStoneBlocks(rect, wall = false) {
  const top = wall ? "#76755d" : "#85846b";
  const side = wall ? "#555845" : "#62634d";
  ctx.fillStyle = side;
  ctx.fillRect(rect.x, rect.y, rect.width, rect.height);
  ctx.fillStyle = top;
  ctx.fillRect(rect.x, rect.y, rect.width, Math.min(8, rect.height));
  ctx.strokeStyle = "rgba(46,50,39,0.32)";
  ctx.lineWidth = 1;
  const blockH = 16;
  const blockW = 34;
  for (let yy = rect.y + 8; yy < rect.y + rect.height; yy += blockH) {
    const row = Math.floor((yy - rect.y) / blockH);
    for (let xx = rect.x + (row % 2 ? -17 : 0); xx < rect.x + rect.width; xx += blockW) {
      ctx.strokeRect(xx, yy, blockW, blockH);
    }
  }
  ctx.fillStyle = "#68864b";
  for (let x = rect.x + 4; x < rect.x + rect.width - 3; x += 18) {
    const h = 3 + ((x * 7) % 7);
    ctx.fillRect(x, rect.y - 2, 14, 4);
    if ((x / 18) % 3 < 1) {
      ctx.fillRect(x + 4, rect.y + 2, 3, h);
    }
  }
}
function drawMangroveWorldDecor() {
  if (currentRoom !== 0) return;
  ctx.save();
  ctx.strokeStyle = "#4c4b36";
  ctx.lineCap = "round";
  const rootZones = [
    90, 1180, 2380, 3440, 4550
  ];
  for (const rootX of rootZones) {
    ctx.lineWidth = 18;
    ctx.beginPath();
    ctx.moveTo(rootX, ground.y + 20);
    ctx.bezierCurveTo(
      rootX - 70, ground.y - 70,
      rootX + 50, ground.y - 170,
      rootX + 24, ground.y - 280
    );
    ctx.stroke();
    ctx.lineWidth = 8;
    for (let i = 0; i < 3; i++) {
      ctx.beginPath();
      ctx.moveTo(rootX, ground.y - 20 - i * 18);
      ctx.quadraticCurveTo(
        rootX + (i - 1) * 58,
        ground.y - 45 - i * 12,
        rootX + (i - 1) * 95,
        ground.y
      );
      ctx.stroke();
    }
  }
  ctx.restore();
  const arches = [520, 2050, 3070, 4380];
  for (const x of arches) {
    ctx.fillStyle = "rgba(93,93,72,0.78)";
    ctx.fillRect(x, 535, 26, 225);
    ctx.fillRect(x + 118, 555, 25, 205);
    ctx.strokeStyle = "rgba(109,108,82,0.85)";
    ctx.lineWidth = 16;
    ctx.beginPath();
    ctx.arc(x + 72, 585, 59, Math.PI, Math.PI * 2);
    ctx.stroke();

    [410, 688], [1010, 578], [1660, 612], [2290, 688],

  ];
  for (const [x, y] of plants) {

      ctx.strokeStyle = i === 0 ? "#b56546" : "#47713f";
