function updateOneMangueEnemy(e, dt) {
  if (!e.alive) return;
  if (e.hurtTimer > 0) e.hurtTimer -= dt;
  if (e.attackCooldownTimer > 0) e.attackCooldownTimer -= dt;
  e.y = ground.y - e.height;
  const enemyCenter = e.x + e.width / 2;
  const macCenter = mac.x + mac.width / 2;
  const distance = Math.abs(macCenter - enemyCenter);
  let speed = e.patrolSpeed;
  if (!mac.dead && distance <= e.detectionRange) {
    e.direction = macCenter < enemyCenter ? -1 : 1;
    speed = e.chaseSpeed;
  } else {
    if (e.x <= e.patrolMin) e.direction = 1;
    if (e.x + e.width >= e.patrolMax) e.direction = -1;
  }
  e.x += e.direction * speed * dt;
  e.x = Math.max(
    e.patrolMin,
    Math.min(e.x, e.patrolMax - e.width)
  );
  if (mac.dead) return;
  const hitbox = {
    x: e.x,
    y: e.y,
    width: e.width,
    height: e.height
  };
  if (
    rectsOverlap(hitbox, getMacHitbox()) &&
    e.attackCooldownTimer <= 0
  ) {
    damageMac(e.damage, e.x + e.width / 2);
    e.attackCooldownTimer = e.attackCooldown;
  }
}

function damageExtraEnemy(e, amount) {
  if (!e || !e.alive) return;
  e.hp -= amount;
  e.hurtTimer = e.hurtDuration;
  if (e.hp <= 0) {
    e.hp = 0;
    e.alive = false;
  }
}
function extraEnemyHitbox(e) {
  return {
    x: e.x,
    y: e.y,
    width: e.width,
    height: e.height
  };
}
function hitFirstExtra(hitbox, amount) {
  if (!hitbox) return false;
  for (const e of getExtraEnemies()) {
    if (!e.alive) continue;
    if (rectsOverlap(hitbox, extraEnemyHitbox(e))) {
      damageExtraEnemy(e, amount);
      return true;
    }
  }
  return false;
}
function updateExtraEnemyCombat() {
  if (
    tongue.active &&
    !grapple.active &&
    !tongue.didHitEnemy
  ) {
    if (hitFirstExtra(getTongueHitbox(), 15)) {
      tongue.didHitEnemy = true;
    }
  }
  if (
    normalAttack.active &&
    normalAttack.hitboxActive &&
    !normalAttack.didHitEnemy
  ) {
    if (hitFirstExtra(getNormalAttackHitbox(), 25)) {
      normalAttack.didHitEnemy = true;
    }
  }
  for (let i = waterShots.length - 1; i >= 0; i--) {
    const shot = waterShots[i];
    const hit = {
      x: shot.x - shot.radius,
      y: shot.y - shot.radius,
      width: shot.radius * 2,
      height: shot.radius * 2
    };
    if (hitFirstExtra(hit, 20)) {
      waterShots.splice(i, 1);
    }
  }
  if (
    waterBurst.active &&
    !waterBurst.didHitEnemy
  ) {
    if (hitFirstExtra(getWaterBurstHitbox(), 35)) {
      waterBurst.didHitEnemy = true;
    }
  }
  if (
    waterUltimate.active &&
    !waterUltimate.didHitEnemy
  ) {
    const hit = {
      x: waterUltimate.waveX,
      y: ground.y - waterUltimate.waveHeight,
      width: waterUltimate.waveWidth,
      height: waterUltimate.waveHeight
    };
    if (hitFirstExtra(hit, 60)) {
      waterUltimate.didHitEnemy = true;
    }
  }
  for (let i = windBlades.length - 1; i >= 0; i--) {
    const blade = windBlades[i];
    const hit = {
      x: blade.x,
      y: blade.y,
      width: blade.width,
      height: blade.height
    };
    if (hitFirstExtra(hit, 20)) {
      windBlades.splice(i, 1);
    }
  }
  if (
    windGust.active &&
    !windGust.didHitEnemy
  ) {
    if (hitFirstExtra(getWindGustHitbox(), 35)) {
      windGust.didHitEnemy = true;
    }
  }
  if (
    windUltimate.active &&
    windUltimate.timer >= windUltimate.spawnDelay &&
    !windUltimate.didHitEnemy
  ) {
    const hit = {
      x: windUltimate.x,
      y: ground.y - windUltimate.height,
      width: windUltimate.width,
      height: windUltimate.height
    };
    if (hitFirstExtra(hit, 70)) {
      windUltimate.didHitEnemy = true;
    }
  }
}

function drawStoneBlocks(rect, wall = false) {
  const top = wall ? "#76755d" : "#85846b";
  const side = wall ? "#555845" : "#62634d";
  ctx.fillStyle = side;
  ctx.fillRect(rect.x, rect.y, rect.width, rect.height);
  ctx.fillStyle = top;
  ctx.fillRect(rect.x, rect.y, rect.width, Math.min(8, rect.height));
  ctx.strokeStyle = "rgba(46,50,39,0.32)";
  ctx.lineWidth = 1;
  const blockH = 16;
  const blockW = 34;
  for (let yy = rect.y + 8; yy < rect.y + rect.height; yy += blockH) {
    const row = Math.floor((yy - rect.y) / blockH);
    for (let xx = rect.x + (row % 2 ? -17 : 0); xx < rect.x + rect.width; xx += blockW) {
      ctx.strokeRect(xx, yy, blockW, blockH);
    }
  }
  ctx.fillStyle = "#68864b";
  for (let x = rect.x + 4; x < rect.x + rect.width - 3; x += 18) {
    const h = 3 + ((x * 7) % 7);
    ctx.fillRect(x, rect.y - 2, 14, 4);
    if ((x / 18) % 3 < 1) {
      ctx.fillRect(x + 4, rect.y + 2, 3, h);
    }
  }
}

    90, 1180, 2380, 3440, 4550
  ];

    ctx.bezierCurveTo(
      rootX - 70, ground.y - 70,
      rootX + 50, ground.y - 170,
      rootX + 24, ground.y - 280
    );
    ctx.stroke();
    ctx.lineWidth = 8;
    for (let i = 0; i < 3; i++) {
      ctx.beginPath();
      ctx.moveTo(rootX, ground.y - 20 - i * 18);
      ctx.quadraticCurveTo(
        rootX + (i - 1) * 58,

let mangueV2SelectedAnchor = null;

function getMangueV2AnchorCandidate() {
  if (currentRoom !== 0) return null;

  const origin = getTongueOrigin();
  let best = null;
  let bestDistance = Infinity;

  for (const anchor of MANGUE_V2_ANCHORS) {
    const d = distance(
      origin.x,
      origin.y,
      anchor.x,
      anchor.y
    );

    if (d > anchor.grabRange) continue;
    if (!tongueLineClearToPoint(anchor.x, anchor.y)) continue;

    const direction =
      anchor.x >= origin.x
        ? 1
        : -1;

    if (
      direction !== mac.facing &&
      d > 185
    ) {
      continue;
    }

    if (d < bestDistance) {
      best = anchor;
      bestDistance = d;
    }
  }

  return best;
}

canGrabPoint = function() {
  if (grapple.active) {
    return true;
  }

  if (currentRoom === 0) {
    const anchor =
      getMangueV2AnchorCandidate();

    if (!anchor) {
      mangueV2SelectedAnchor = null;
      return false;
    }

    mangueV2SelectedAnchor = anchor;

    grapplePoint.x = anchor.x;
    grapplePoint.y = anchor.y;
    grapplePoint.radius = anchor.radius;
    grapplePoint.grabRange = anchor.grabRange;

    return true;
  }

  const origin = getTongueOrigin();

  return (
    distance(
      origin.x,
      origin.y,
      grapplePoint.x,
      grapplePoint.y
    ) <= grapplePoint.grabRange &&
    tongueLineClearToPoint(
      grapplePoint.x,
      grapplePoint.y
    )
  );
};

resetMangueEnemies = function(room) {
  for (const e of mangueEnemies) {
    if (e.room !== room) continue;

    if (typeof e.surfaceY !== "number") {
      e.surfaceY = ground.y;
    }

    e.x = e.spawnX;
    e.y = e.surfaceY - e.height;
    e.hp = e.maxHp;
    e.alive = true;
    e.attackCooldownTimer = 0;
    e.hurtTimer = 0;
  }
};

updateOneMangueEnemy = function(e, dt) {
  if (!e.alive) return;

  if (typeof e.surfaceY !== "number") {
    e.surfaceY = ground.y;
  }

  if (e.hurtTimer > 0) {
    e.hurtTimer -= dt;
  }

  if (e.attackCooldownTimer > 0) {
    e.attackCooldownTimer -= dt;
  }

  e.y =
    e.surfaceY -
    e.height;

  const enemyCenter =
    e.x +
    e.width / 2;

  const macCenter =
    mac.x +
    mac.width / 2;

  const distanceToMac =
    Math.abs(
      macCenter -
      enemyCenter
    );

  const verticalDistance =
    Math.abs(
      (
        mac.y +
        mac.height / 2
      ) -
      (
        e.y +
        e.height / 2
      )
    );

  let speed =
    e.patrolSpeed;

  if (
    !mac.dead &&
    distanceToMac <= e.detectionRange &&
    verticalDistance < 95
  ) {
    e.direction =
      macCenter < enemyCenter
        ? -1
        : 1;

    speed =
      e.chaseSpeed;
  } else {
    if (e.x <= e.patrolMin) {
      e.direction = 1;
    }

    if (
      e.x +
      e.width >=
      e.patrolMax
    ) {
      e.direction = -1;
    }
  }

  e.x +=
    e.direction *
    speed *
    dt;

  e.x =
    Math.max(
      e.patrolMin,
      Math.min(
        e.x,
        e.patrolMax -
        e.width
      )
    );

  if (mac.dead) return;

  const hitbox = {
    x: e.x,
    y: e.y,
    width: e.width,
    height: e.height
  };

  if (
    rectsOverlap(
      hitbox,
      getMacHitbox()
    ) &&
    e.attackCooldownTimer <= 0
  ) {
    damageMac(
      e.damage,
      e.x +
      e.width / 2
    );

    e.attackCooldownTimer =
      e.attackCooldown;
  }
};

function drawMangueV2Pit(
  pit
) {
  const t =
    performance.now() *
    0.001;

  ctx.fillStyle =
    "#447f80";

  ctx.fillRect(
    pit.x,
    ground.y,
    pit.width,
    WORLD_HEIGHT -
      ground.y
  );

  ctx.strokeStyle =
    "rgba(222,244,219,0.52)";

  ctx.lineWidth = 2;

  for (
    let y =
      ground.y + 10;
    y <
      WORLD_HEIGHT;
    y += 24
  ) {
    const phase =
      Math.sin(
        t * 1.8 +
        y * 0.03
      ) * 10;

    for (
      let x =
        pit.x + 8;
      x <
        pit.x +
        pit.width - 20;
      x += 58
    ) {
      ctx.beginPath();

      ctx.moveTo(
        x + phase,
        y
      );

      ctx.quadraticCurveTo(
        x + 17 + phase,
        y - 3,
        x + 35 + phase,
        y
      );

      ctx.stroke();
    }
  }

  ctx.strokeStyle =
    "#4d4b36";

  ctx.lineWidth = 6;

  for (
    let i = 0;
    i < 3;
    i++
  ) {
    const x =
      pit.x +
      pit.width *

        0.22 +
        i * 0.27

    ctx.closePath();
    ctx.stroke();
  }
};

function drawMangueV2Anchor(
  anchor,
  reachable
) {
  ctx.save();

  ctx.strokeStyle =
    anchor.kind === "vine"
      ? "#53633d"
      : "#554832";

  ctx.lineCap = "round";

  ctx.lineWidth =
    anchor.kind === "vine"
      ? 6
      : 10;

  ctx.beginPath();

  ctx.moveTo(
    anchor.x -
      (
        anchor.kind === "vine"
          ? 0
          : 52
      ),
    anchor.y - 62
  );

  ctx.quadraticCurveTo(
    anchor.x - 8,
    anchor.y - 28,
    anchor.x,
    anchor.y
  );

  ctx.stroke();

  ctx.fillStyle =
    reachable
      ? "#cbdc88"
      : "#8fa064";

  ctx.beginPath();

  ctx.ellipse(
    anchor.x,
    anchor.y,
    11,
    8,
    -0.2,
    0,
    Math.PI * 2
  );

  ctx.fill();

  ctx.strokeStyle =
    "rgba(237,245,194,0.65)";

  ctx.lineWidth = 2;
  ctx.stroke();

  ctx.strokeStyle =
    "#668b51";

  ctx.lineWidth = 3;

  for (
    let i = -2;
    i <= 2;
    i++
  ) {
    ctx.beginPath();

    ctx.moveTo(
      anchor.x,
      anchor.y - 7
    );

    ctx.quadraticCurveTo(
      anchor.x +
        i * 4,
      anchor.y - 17,
      anchor.x +
        i * 9,
      anchor.y - 23
    );

    ctx.stroke();
  }

  // Só o primeiro ponto ensina o botão; depois o cenário fala sozinho.
  if (
    anchor.id === "raiz_ponte_1" &&
    mac.x < 820
  ) {
    ctx.fillStyle =
      "rgba(35,49,36,0.72)";

    ctx.beginPath();

    ctx.arc(
      anchor.x - 18,
      anchor.y - 40,
      13,
      0,
      Math.PI * 2
    );

    ctx.fill();

    ctx.fillStyle =
      "#f1f0cf";

    ctx.font =
      "bold 14px sans-serif";

    ctx.textAlign =
      "center";

    ctx.textBaseline =
      "middle";

    ctx.fillText(
      "C",
      anchor.x - 18,
      anchor.y - 40
    );
  }

  ctx.restore();
}

drawGrapplePoint = function() {
  if (currentRoom !== 0) {
    const origin =
      getTongueOrigin();

    const reachable =
      distance(
        origin.x,
        origin.y,
        grapplePoint.x,
        grapplePoint.y
      ) <=
        grapplePoint.grabRange &&
      tongueLineClearToPoint(
        grapplePoint.x,
        grapplePoint.y
      );

    ctx.save();

    ctx.fillStyle =
      reachable
        ? "#5eead4"
        : "#38bdf8";

    ctx.beginPath();

    ctx.arc(
      grapplePoint.x,
      grapplePoint.y,
      grapplePoint.radius,
      0,
      Math.PI * 2
    );

    ctx.fill();

    ctx.strokeStyle =
      "#dffcff";

    ctx.lineWidth = 3;
    ctx.stroke();

    ctx.restore();
    return;
  }

  const origin =
    getTongueOrigin();

  for (
    const anchor of
    MANGUE_V2_ANCHORS
  ) {
    const d =
      distance(
        origin.x,
        origin.y,
        anchor.x,
        anchor.y
      );

    const direction =
      anchor.x >= origin.x
        ? 1
        : -1;

    const reachable =
      !grapple.active &&
      d <= anchor.grabRange &&
      (
        direction === mac.facing ||
        d <= 185
      ) &&
      tongueLineClearToPoint(
        anchor.x,
        anchor.y
      );

    drawMangueV2Anchor(
      anchor,
      reachable
    );
  }
};

  ctx.lineCap = "round";

  for (

    "#6b6d56";

drawWorldGeometry = function() {

    for (
      const segment of
      MANGUE_V2_GROUND_SEGMENTS
    ) {
      drawMangueV2GroundSegment(
        segment
      );
    }

    for (
      const pit of
      MANGUE_V2_PITS
    ) {
      drawMangueV2Pit(
        pit
      );
    }
  } else {
    ctx.fillStyle =
      "#5c5c5c";

    ctx.fillRect(
      ground.x,
      ground.y,
      ground.width,
      ground.height
    );
  }

  for (
    const platform of
    platforms
  ) {
    if (currentRoom === 0) {
      drawStoneBlocks(
        platform,
        false
      );
    } else {
      ctx.fillStyle =
        "#777777";

      ctx.fillRect(
        platform.x,
        platform.y,
        platform.width,
        platform.height
      );
    }
  }

  for (
    const wall of walls
  ) {
    if (currentRoom === 0) {
      drawStoneBlocks(
        wall,
        true
      );
    } else {
      ctx.fillStyle =
        "#595959";

      ctx.fillRect(
        wall.x,
        wall.y,
        wall.width,
        wall.height
      );
    }
  }

  if (DEBUG_HITBOX) {
    ctx.save();

    ctx.strokeStyle =
      "rgba(255,255,255,0.28)";

    for (
      const solid of
      solids
    ) {
      ctx.strokeRect(
        solid.x,
        solid.y,
        solid.width,
        solid.height
      );
    }

    ctx.restore();
  }
};
